cmake_minimum_required(VERSION 3.24)
project(objc3_native_linux_runtime LANGUAGES C CXX OBJC OBJCXX)
include(CTest)
get_filename_component(OBJC3_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${OBJC3_REPO}/native/objc3c/cmake/Objc3NativeRuntime.cmake")
objc3c_configure_native_runtime(PROFILE linux-x86_64-libobjc2)
objc3c_add_native_object_adapter(TARGET objc3_native_adapter)
set(_fixtures "${OBJC3_REPO}/tests/tooling/fixtures/clang_foundation/runtime")
foreach(_case IN ITEMS root_class_arc_weak_blocks root_class_arc_consumed root_class_mrc root_class_allocation
    root_class_objcxx_aggregate_cleanup native_weak_slot_adapter)
  if(_case MATCHES "^(root_class_objcxx|native_weak)")
    set(_extension mm)
  else()
    set(_extension m)
  endif()
  add_executable("${_case}" "${_fixtures}/${_case}.${_extension}")
  target_compile_features("${_case}" PRIVATE cxx_std_20)
  target_compile_options("${_case}" PRIVATE -fblocks)
  if(_case STREQUAL "root_class_mrc")
    target_compile_options("${_case}" PRIVATE -fno-objc-arc)
  else()
    target_compile_options("${_case}" PRIVATE -fobjc-arc)
  endif()
  target_link_libraries("${_case}" PRIVATE objc3_native_adapter)
  add_test(NAME "${_case}" COMMAND "${_case}")
endforeach()

objc3c_install_native_runtime(TARGET objc3_native_adapter DESTINATION ".")

add_executable(exception_adapter "${_fixtures}/exception_adapter.m" "${_fixtures}/exception_adapter.mm")
target_compile_features(exception_adapter PRIVATE cxx_std_20)
target_compile_options(exception_adapter PRIVATE -fobjc-arc -fblocks -fexceptions -fobjc-arc-exceptions)
target_link_libraries(exception_adapter PRIVATE objc3_native_adapter)
add_test(NAME exception_adapter COMMAND exception_adapter)
